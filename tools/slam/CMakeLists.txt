# pibridge-slam: offline 6DoF tools (port of omarchyvr's `omarchyvr-slam`; today the `export` subcommand: the Basalt
# calibration, VIO config and Monado SLAM_CONFIG toml). Output files and text identical to the Rust tool's.

# Embedded data, like the Rust include_str!: Basalt's EuRoC VIO config (BSD-3-Clause, see NOTICE) and the Dream Air
# reference gyro bias (devices/dream-air/reference/gyro-bias, no serials).
set(pibridge_slam_embed "${CMAKE_CURRENT_BINARY_DIR}/embedded.cpp")
set(pibridge_slam_base "${CMAKE_CURRENT_SOURCE_DIR}/data/basalt-euroc_config.json")
set(pibridge_slam_gyro "${PIBRIDGE_SOURCE_DEVICES_DIR}/dream-air/reference/gyro-bias")
function(pibridge_slam_bytes file out_var len_var)
    file(READ "${file}" hex HEX)
    string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," bytes "${hex}")
    string(LENGTH "${hex}" n)
    math(EXPR n "${n} / 2")
    set(${out_var} "${bytes}" PARENT_SCOPE)
    set(${len_var} "${n}" PARENT_SCOPE)
endfunction()
pibridge_slam_bytes("${pibridge_slam_base}" base_bytes base_len)
pibridge_slam_bytes("${pibridge_slam_gyro}" gyro_bytes gyro_len)
file(CONFIGURE OUTPUT "${pibridge_slam_embed}" CONTENT
"// SPDX-License-Identifier: MIT
// Generated by tools/slam/CMakeLists.txt from tools/slam/data/basalt-euroc_config.json (Basalt, BSD-3-Clause) and
// devices/dream-air/reference/gyro-bias.
#include \"export.h\"
namespace pibridge::slamtool {
namespace {
const char k_base[] = {@base_bytes@ 0};
const char k_gyro[] = {@gyro_bytes@ 0};
}
std::string_view builtin_base() { return std::string_view(k_base, @base_len@); }
std::string_view reference_gyro_bias() { return std::string_view(k_gyro, @gyro_len@); }
}
" @ONLY)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${pibridge_slam_base}" "${pibridge_slam_gyro}")

add_library(pibridge_slamtool STATIC src/export.cpp src/headreplay.cpp ${pibridge_slam_embed})
target_include_directories(pibridge_slamtool PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_link_libraries(pibridge_slamtool PUBLIC pibridge_cam)
target_compile_options(pibridge_slamtool PRIVATE -ffp-contract=off -fwrapv)

add_executable(pibridge-slam src/main.cpp)
target_link_libraries(pibridge-slam PRIVATE pibridge_slamtool)
target_compile_options(pibridge-slam PRIVATE -ffp-contract=off -fwrapv)
install(TARGETS pibridge-slam RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})

if(PIBRIDGE_BUILD_TESTS)

    target_link_libraries(pibridge_slam_test_export PRIVATE pibridge_slamtool pibridge_common)

    set_tests_properties(pibridge_slam_export PROPERTIES ENVIRONMENT "TMPDIR=${CMAKE_CURRENT_BINARY_DIR}")
endif()
